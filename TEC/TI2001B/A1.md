## MySQL permissons

```mysql
CREATE DATABASE cars_accidents;
CREATE USER 'kopepod'@'localhost' IDENTIFIED BY '<pass>';
GRANT ALL PRIVILEGES ON cars_accidents.* TO 'kopepod'@'localhost';
GRANT SYSTEM_VARIABLES_ADMIN, SUPER ON *.* TO 'kopepod'@'localhost';
GRANT ALL PRIVILEGES ON *.* TO 'kopepod'@'localhost' WITH GRANT OPTION;
FLUSH PRIVILEGES;
EXIT;
```

## Create DB from CSV

```mysql
SET GLOBAL local_infile=1;
create database fisher;
use fisher;

CREATE TABLE General (sepal_length FLOAT, sepal_width FLOAT, petal_length FLOAT, petal_width FLOAT, variety VARCHAR(30), reg_date TIMESTAMP);

```

```bash
mysql -u kopepod -p --local_infile=1 fisher -e "LOAD DATA LOCAL INFILE 'fisher_iris.csv' INTO TABLE General FIELDS TERMINATED BY ',' LINES TERMINATED BY '\n'"
```

```r
library(DBI)
library(RMySQL)


con <- dbConnect(
  RMySQL::MySQL(),
  user     = "kopepod",
  password = "5tr0ngPa55",
  dbname   = "fisher",
  host     = "127.0.0.1",          # Your MySQL server IP
  port     = 3306                  # Default MySQL port
)

query <- "SELECT sepal_length,sepal_width FROM General;"

df <- dbGetQuery(con, query)

head(df)


ggplot(df, aes(x = sepal_length, y = sepal_width)) +
  geom_point() +
  labs(x = "sepal_length", y = "sepal_width", title = "Sepal")
```
